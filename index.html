const standingsUrl = 'https://site.api.espn.com/apis/v2/sports/baseball/mlb/standings'; async function loadStandings() { try { const response = await fetch(standingsUrl); const data = await response.json(); const children = data.children; const alCentral = children.find(group => group.name.includes('American League Central') ); const nlCentral = children.find(group => group.name.includes('National League Central') ); renderDivision('al-central-body', alCentral.standings.entries); renderDivision('nl-central-body', nlCentral.standings.entries); const now = new Date(); document.getElementById('updated-time').innerText = `Last Updated: ${now.toLocaleString()}`; } catch (error) { console.error(error); document.getElementById('al-central-body').innerHTML = 'Failed to load standings'; document.getElementById('nl-central-body').innerHTML = 'Failed to load standings'; } } function renderDivision(elementId, teams) { const tbody = document.getElementById(elementId); tbody.innerHTML = ''; teams.forEach(team => { const stats = {}; team.stats.forEach(stat => { stats[stat.name] = stat.displayValue; }); const tr = document.createElement('tr'); tr.innerHTML = ` ${team.team.displayName} ${stats.wins || '-'} ${stats.losses || '-'} ${stats.gamesBehind || '0'} `; tbody.appendChild(tr); }); } loadStandings(); setInterval(loadStandings, 300000);
